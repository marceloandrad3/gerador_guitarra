     1	#!/usr/bin/env python3
     2	"""
     3	Auditor JGuitar - compara shapes do NOSSO gerador com os do JGuitar.
     4	
     5	Para cada acorde do universo (12 tonicas x 15 tipos = 180 acordes):
     6	  - Gera shapes com nosso sistema (gerar_dinamico)
     7	  - Busca shapes no JGuitar (chordsearch)
     8	  - Registra em SQLite (auditoria.db):
     9	      match       = presente nos dois
    10	      so_nosso    = so nosso sistema gerou (precisa 2a fonte para validar)
    11	      so_jguitar  = JGuitar tem e NAO geramos (FALHA de cobertura)
    12	
    13	Formato de shape (ambos): 6 campos separados por virgula, indice 0 = 6a corda.
    14	Exemplo: X,3,2,0,1,0
    15	"""
    16	
    17	import datetime
    18	import re
    19	import sqlite3
    20	import sys
    21	import time
    22	import ssl
    23	import urllib.parse
    24	import urllib.request
    25	
    26	# macOS: Python oficial nem sempre tem certificados instalados.
    27	# Usamos o bundle do certifi diretamente (nao depende do sistema).
    28	try:
    29	    import certifi
    30	    SSL_CTX = ssl.create_default_context(cafile=certifi.where())
    31	except ImportError:
    32	    SSL_CTX = None
    33	
    34	from gerador_web_dinamico import gerar_dinamico, NOTAS
    35	
    36	DB_PATH = "auditoria.db"
    37	BASE_URL = "https://jguitar.com/chordsearch?chordsearch="
    38	PAUSA_SEGUNDOS = 1.0
    39	
    40	# Nosso tipo -> sufixo da query no JGuitar (mapeamento validado em teste)
    41	TIPO_JGUITAR = {
    42	    "":     "",
    43	    "m":    "m",
    44	    "7":    "7",
    45	    "m7":   "m7",
    46	    "7M":   "maj7",
    47	    "m7M":  "mmaj7",
    48	    "dim":  "dim",
    49	    "dim7": "dim7",
    50	    "aug":  "aug",
    51	    "sus2": "sus2",
    52	    "sus4": "sus4",
    53	    "6":    "6",
    54	    "m6":   "m6",
    55	    "9":    "9",
    56	    "add9": "add9",
    57	}
    58	
    59	
    60	def acorde_para_query(acorde):
    61	    """Converte nome interno (ex: C7M, F#m7M) em query JGuitar (ex: Cmaj7, F#mmaj7)."""
    62	    m = re.match(r"^([A-G][#b]?)(.*)$", acorde)
    63	    if not m:
    64	        return acorde
    65	    tonica, tipo = m.group(1), m.group(2)
    66	    return tonica + TIPO_JGUITAR.get(tipo, tipo)
    67	
    68	
    69	# Shape = ultimo segmento do nome do PNG: ...-{shape}.png
    70	# Shape = 6 campos (digito ou x) separados por %2C
    71	PADRAO_SHAPE = r'/images/chordshape/[^"]*?-(?:\d+|[xX])(?:%2C(?:\d+|[xX])){5}\.png'
    72	
    73	
    74	def universo_acordes():
    75	    """12 tonicas x 15 tipos = 180 acordes."""
    76	    for tonica in NOTAS:
    77	        for tipo in TIPO_JGUITAR:
    78	            yield tonica + tipo
    79	
    80	
    81	def buscar_jguitar(acorde):
    82	    """Retorna SET de shapes que o JGuitar publica para o acorde."""
    83	    url = BASE_URL + urllib.parse.quote(acorde, safe="")
    84	    req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
    85	    html = urllib.request.urlopen(req, timeout=20, context=SSL_CTX).read().decode("utf-8", errors="replace")
    86	    shapes = set()
    87	    for bruto in re.findall(PADRAO_SHAPE, html):
    88	        segmento = bruto.rsplit("/", 1)[-1]   # nome do arquivo
    89	        segmento = segmento.rsplit("-", 1)[-1]  # apos o ultimo traco
    90	        segmento = segmento[:-4]                # remove .png
    91	        shapes.add(urllib.parse.unquote(segmento).upper())
    92	    return shapes
    93	
    94	
    95	def shapes_nossos(acorde):
    96	    """Retorna SET de shapes que o nosso gerador produz."""
    97	    resultado = gerar_dinamico(acorde)
    98	    return {
    99	        ",".join(str(x).upper() for x in d["diagrama"])
   100	        for d in resultado["diagramas"]
   101	    }
   102	
   103	
   104	def criar_banco():
   105	    con = sqlite3.connect(DB_PATH)
   106	    con.execute("""
   107	        CREATE TABLE IF NOT EXISTS auditoria (
   108	            id INTEGER PRIMARY KEY AUTOINCREMENT,
   109	            acorde TEXT NOT NULL,
   110	            shape TEXT NOT NULL,
   111	            gerado_por TEXT NOT NULL,
   112	            status TEXT NOT NULL,
   113	            validacao_extra TEXT,
   114	            fonte_extra TEXT,
   115	            data TEXT NOT NULL,
   116	            UNIQUE(acorde, shape)
   117	        )
   118	    """)
   119	    con.commit()
   120	    return con
   121	
   122	
   123	def registrar(con, acorde, shape, gerado_por, status, agora):
   124	    con.execute("""
   125	        INSERT INTO auditoria (acorde, shape, gerado_por, status, data)
   126	        VALUES (?, ?, ?, ?, ?)
   127	        ON CONFLICT(acorde, shape) DO UPDATE SET
   128	            gerado_por = excluded.gerado_por,
   129	            status = excluded.status,
   130	            data = excluded.data
   131	    """, (acorde, shape, gerado_por, status, agora))
   132	
   133	
   134	def auditar_conjunto(con, agora, acordes):
   135	    totais = {"acordes": 0, "match": 0, "so_nosso": 0, "so_jguitar": 0, "falhas": []}
   136	    for acorde in acordes:
   137	        totais["acordes"] += 1
   138	        nosso = shapes_nossos(acorde)
   139	        jg = buscar_jguitar(acorde_para_query(acorde))
   140	
   141	        for s in nosso & jg:
   142	            registrar(con, acorde, s, "ambos", "match", agora)
   143	        for s in nosso - jg:
   144	            registrar(con, acorde, s, "nosso", "so_nosso", agora)
   145	        for s in jg - nosso:
   146	            registrar(con, acorde, s, "jguitar", "so_jguitar", agora)
   147	        con.commit()
   148	
   149	        so_jg = len(jg - nosso)
   150	        totais["match"] += len(nosso & jg)
   151	        totais["so_nosso"] += len(nosso - jg)
   152	        totais["so_jguitar"] += so_jg
   153	        if so_jg:
   154	            totais["falhas"].append((acorde, so_jg))
   155	
   156	        print(f"[{totais['acordes']:>3}] {acorde:<8} "
   157	              f"nosso={len(nosso):>2} jguitar={len(jg):>2} "
   158	              f"match={len(nosso & jg):>2} so_nosso={len(nosso - jg):>2} "
   159	              f"so_jguitar={so_jg:>2}")
   160	        time.sleep(PAUSA_SEGUNDOS)
   161	    return totais
   162	
   163	
   164	def auditar(acordes=None):
   165	    con = criar_banco()
   166	    agora = datetime.datetime.now().isoformat(timespec="seconds")
   167	    if acordes is None:
   168	        acordes = universo_acordes()
   169	    totais = auditar_conjunto(con, agora, acordes)
   170	    con.close()
   171	
   172	    print("=" * 60)
   173	    print(f" Acordes auditados:          {totais['acordes']}")
   174	    print(f" Shapes em comum (match):    {totais['match']}")
   175	    print(f" So nosso (validar depois):  {totais['so_nosso']}")
   176	    print(f" So JGuitar (FALHA nossa):   {totais['so_jguitar']}")
   177	    if totais["falhas"]:
   178	        print("-" * 60)
   179	        print(" Acordes com voicings que NAO geramos:")
   180	        for acorde, qtd in totais["falhas"]:
   181	            print(f"   {acorde}: {qtd} voicing(s) ausente(s)")
   182	    print("=" * 60)
   183	
   184	
   185	if __name__ == "__main__":
   186	    # Sem argumentos: audita os 180 acordes
   187	    # Com argumentos: audita apenas os acordes informados (modo teste)
   188	    auditar(sys.argv[1:] if len(sys.argv) > 1 else None)

### validacao.py (importado por gerador_web_dinamico.py)
########################################################
     1	"""
     2	Modulo de validacao de shapes de acordes.
     3	
     4	Regra (definicao de acorde): um shape so e valido se
     5	1. Nenhuma nota errada: toda nota tocada pertence ao acorde.
     6	2. Nenhuma nota faltando: todos os intervalos do acorde estao presentes,
     7	   exceto os explicitamente tolerados via `tolerar_faltando` (ausencia da
     8	   5a justa - pratica convencional do violao, Wikipedia "Guitar chord":
     9	   "the fifth is often omitted"). Tonica e 5as alteradas nunca toleradas.
    10	"""
    11	
    12	
    13	def _parse_casas(diagrama):
    14	    """Converte valores do diagrama ('X', '0', '3'...) em casa int ou None."""
    15	    casas = []
    16	    for valor in diagrama:
    17	        try:
    18	            casa = int(valor)
    19	        except (TypeError, ValueError):
    20	            casa = None
    21	        casas.append(casa if casa is not None and casa >= 0 else None)
    22	    return casas
    23	
    24	
    25	def notas_tocadas(casas, afinacao):
    26	    """Classes de altura (0-11) das cordas tocadas (inclui cordas soltas)."""
    27	    return {(afinacao[i] + c) % 12 for i, c in enumerate(casas) if c is not None}
    28	
    29	
    30	def shape_valido(casas, afinacao, notas_acorde, tolerar_faltando=frozenset()):
    31	    """Valida o shape contra as notas do acorde.
    32	
    33	    Sem sobra sempre; sem falta exceto notas em `tolerar_faltando`
    34	    (classes de altura cuja ausencia e aceita, ex.: 5a justa).
    35	    """
    36	    tocadas = notas_tocadas(casas, afinacao)
    37	    return (tocadas.issubset(notas_acorde)
    38	            and notas_acorde.issubset(tocadas | set(tolerar_faltando)))
    39	
    40	
    41	def diagrama_valido(diagrama, afinacao, notas_acorde, tolerar_faltando=frozenset()):
    42	    """Valida um diagrama (lista de strings) contra as notas do acorde."""
    43	    return shape_valido(_parse_casas(diagrama), afinacao, notas_acorde,
    44	                        tolerar_faltando)
